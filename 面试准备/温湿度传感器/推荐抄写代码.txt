==================== AHT_Init() 代码 ====================

uint8_t AHT_Init(void)
{
    AHT_CLK_ENABLE;                 // 使能 GPIOB 时钟
    IICInit(&AHT_bus);              // 初始化 I2C 引脚为推挽输出

    delay_ms(40);                   // 等待传感器上电稳定

    // 检查校准使能位，未校准则发送初始化命令
    if(AHT_Read_Status()&&0x08!=0x08)
    {
        IICStart(&AHT_bus);
        IICSendByte(&AHT_bus,0x70);     // 写地址 0x38 << 1 | 0
        IICWaitAck(&AHT_bus);
        IICSendByte(&AHT_bus,0xBE);     // 触发校准命令
        IICWaitAck(&AHT_bus);
        IICSendByte(&AHT_bus,0x08);     // 参数
        IICWaitAck(&AHT_bus);
        IICSendByte(&AHT_bus,0x00);     // 参数
        IICWaitAck(&AHT_bus);
        IICStop(&AHT_bus);
        delay_ms(10);
    }

    return 0;
}


==================== AHT_Read() 代码 ====================

uint8_t AHT_Read(float *humi, float *temp)
{
    uint8_t cnt=5;
    uint8_t  Byte_1th=0;
    uint8_t  Byte_2th=0;
    uint8_t  Byte_3th=0;
    uint8_t  Byte_4th=0;
    uint8_t  Byte_5th=0;
    uint8_t  Byte_6th=0;
    uint32_t RetuData = 0;

    // ---------- 第一步：触发测量 ----------
    IICStart(&AHT_bus);
    IICSendByte(&AHT_bus,0x70);      // 写地址
    IICWaitAck(&AHT_bus);
    IICSendByte(&AHT_bus,0xAC);      // 触发测量命令
    IICWaitAck(&AHT_bus);
    IICSendByte(&AHT_bus,0x33);      // 参数1
    IICWaitAck(&AHT_bus);
    IICSendByte(&AHT_bus,0x00);      // 参数2
    IICWaitAck(&AHT_bus);
    IICStop(&AHT_bus);

    // ---------- 第二步：等待测量完成 ----------
    delay_ms(80);                    // 等待至少 80ms
    while(AHT_Read_Status()&0x80==0x80 && cnt)  // bit7=1 表示忙
    {
        delay_ms(5);
        cnt--;
        AHT_Read_Status();
    }
    if(!cnt)
    {return 1;}

    // ---------- 第三步：读取 6 字节数据 ----------
    IICStart(&AHT_bus);
    IICSendByte(&AHT_bus,0x71);      // 读地址
    IICWaitAck(&AHT_bus);
    Byte_1th = IICReceiveByte(&AHT_bus);  // 状态字节
    IICSendAck(&AHT_bus);
    Byte_2th = IICReceiveByte(&AHT_bus);  // 湿度 Byte2
    IICSendAck(&AHT_bus);
    Byte_3th = IICReceiveByte(&AHT_bus);  // 湿度 Byte3
    IICSendAck(&AHT_bus);
    Byte_4th = IICReceiveByte(&AHT_bus);  // 湿度 Byte4 + 温度 Byte1
    IICSendAck(&AHT_bus);
    Byte_5th = IICReceiveByte(&AHT_bus);  // 温度 Byte2
    IICSendAck(&AHT_bus);
    Byte_6th = IICReceiveByte(&AHT_bus);  // 温度 Byte3
    IICSendNotAck(&AHT_bus);              // 最后一位发 NACK
    IICStop(&AHT_bus);

    // ---------- 第四步：计算湿度 ----------
    // 湿度：Byte2[7:0] + Byte3[7:0] + Byte4[7:4]，共 20 位
    RetuData = (RetuData|Byte_2th)<<8;
    RetuData = (RetuData|Byte_3th)<<8;
    RetuData = (RetuData|Byte_4th);
    RetuData = RetuData >> 4;                      // 去掉 Byte4 低 4 位
    *humi = (RetuData * 1000 >> 20);              // *100 / 2^20
    *humi /= 10;

    // ---------- 第五步：计算温度 ----------
    // 温度：Byte4[3:0] + Byte5[7:0] + Byte6[7:0]，共 20 位
    RetuData = 0;
    RetuData = (RetuData|(Byte_4th&0x0f))<<8;    // 取 Byte4 低 4 位
    RetuData = (RetuData|Byte_5th)<<8;
    RetuData = (RetuData|Byte_6th);
    RetuData = RetuData & 0xfffff;                // 确保 20 位
    *temp = ((RetuData * 2000 >> 20) - 500);     // *200/2^20 - 50
    *temp /= 10;

    return 0;
}
